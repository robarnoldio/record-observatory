---
import fs from 'node:fs';
import path from 'node:path';
import { parse } from 'csv-parse/sync';


// Build one static Astro route for every curated reference JSON file.
//
// For example:
//   reference/herbie-hancock.json
// becomes:
//   /artists/herbie-hancock
//
// The reference filename therefore serves as the URL slug.
export function getStaticPaths() {
  const referenceDir = path.resolve('reference');

  return fs
    .readdirSync(referenceDir)
    .filter(name => name.endsWith('.json'))
    .map(name => ({
      params: {
        artist: name.replace(/\.json$/, '')
      }
    }));
}


// Get the artist slug supplied by the dynamic route.
//
// For:
//   /artists/herbie-hancock
//
// Astro.params.artist is:
//   "herbie-hancock"
const { artist } = Astro.params;


// Convert the URL slug to a temporary display name.
//
// Examples:
//   herbie-hancock -> Herbie Hancock
//   weather-report -> Weather Report
//
// This is intentionally temporary. The reference data will eventually
// carry canonical artist metadata so names such as "Antônio Carlos Jobim"
// do not need to be inferred from filenames.
const artistName = artist
  .split('-')
  .map(word => word.charAt(0).toUpperCase() + word.slice(1))
  .join(' ');


// Load the curated reference discography associated with this route.
//
// The reference file defines which records belong in this artist
// collection. Its array order is also meaningful and is preserved
// through reconciliation and display.
const referencePath = path.resolve(`reference/${artist}.json`);

const artistReference = JSON.parse(
  fs.readFileSync(referencePath, 'utf8')
);


// Find the most recent Discogs collection export.
//
// Discogs export filenames contain sortable timestamps, so sorting the
// matching filenames lets the final entry represent the newest export.
const collectionDir = path.resolve('data/collection');

const csvFiles = fs
  .readdirSync(collectionDir)
  .filter(name => /^Robarnold-collection-.*\.csv$/.test(name))
  .sort();

if (csvFiles.length === 0) {
  throw new Error('No Discogs collection CSV found');
}

const csvFilename = csvFiles.at(-1);
const csvPath = path.join(collectionDir, csvFilename);


// Parse the Discogs CSV into an array of JavaScript objects.
//
// With columns: true, each CSV row becomes an object whose property
// names come from the CSV header row.
const csvText = fs.readFileSync(csvPath, 'utf8');

const collection = parse(csvText, {
  columns: true,
  skip_empty_lines: true,
});


// Restrict reconciliation candidates to LP records.
//
// We deliberately DO NOT filter on the Discogs Artist field here.
// Membership in an artist collection is defined by the curated reference
// JSON rather than by Discogs artist attribution. This permits intentional
// cases such as a sideman or collaboration album being included in another
// artist's curated collection.
//
// Discogs expresses multi-disc LP sets as values such as "2xLP", so both
// "LP" and multi-LP format tokens are accepted.
const lpCollection = collection.filter(record =>
  record.Format?.split(',').some(x => /^\d*xLP$|^LP$/.test(x.trim()))
);


// Handy diagnostic probe. Change the search string and uncomment
// console.log() when investigating a particular Discogs collection row.
//
// This is server-side Astro frontmatter, so console output appears in the
// terminal running the Astro development server rather than the browser.
const debugRecords = collection.filter(record =>
  record.Title?.toLowerCase().includes('weather report')
);

console.log(debugRecords);

// Normalize titles for matching.
//
// Matching is deliberately conservative for now. Capitalization and
// surrounding whitespace are ignored, but otherwise the titles must be
// identical. We prefer fixing curatorial issues in the reference data
// rather than hiding them behind increasingly permissive fuzzy matching.
function normalizeTitle(title) {
  return title
    .toLowerCase()
    .trim();
}


// Reconcile every curated reference entry against all owned LPs.
//
// Each reference entry is copied into a result object and supplemented with:
//   owned  - true when at least one matching physical LP is found
//   copies - all matching Discogs collection rows
//
// Known limitation:
// Different conceptual albums having exactly the same title cannot yet be
// distinguished. Examples discovered during testing include the multiple
// "The Best Of Herbie Hancock" and "Weather Report" albums. These will
// eventually require explicit matching metadata in the reference data.
const reconciliation = artistReference.map(reference => {
  const matches = lpCollection.filter(record =>
    normalizeTitle(record.Title) === normalizeTitle(reference.title)
  );

  return {
    ...reference,
    owned: matches.length > 0,
    copies: matches
  };
});


// Produce simple completeness totals for the page.
const ownedCount = reconciliation.filter(record => record.owned).length;
const missingCount = reconciliation.length - ownedCount;
---

<html lang="en">
  <head>
    <meta charset="utf-8" />
    <title>{artistName} Reconciliation Test</title>
  </head>

  <body>
    <h1>{artistName} Reconciliation Test</h1>

    <p>Discogs export: {csvFilename}</p>

    <h2>Input counts</h2>

    <ul>
      <li>Reference records in {artistName} discography: {artistReference.length}</li>
      <li>Total discogs export records: {collection.length}</li>
      <li>Collection records of LP type: {lpCollection.length}</li>
    </ul>

    <h2>Completeness report</h2>

    <p>
Owned: {ownedCount} / {reconciliation.length} ({((100 * ownedCount) / reconciliation.length).toFixed(2)}% complete)
<br />      Missing: {missingCount}
    </p>

    <h2>Reconciliation</h2>
    
    <table>
      <thead>
        <tr>
          <th>Year</th>
          <th>Title</th>
          <th>Status</th>
          <th>Copies</th>
        </tr>
      </thead>

      <tbody>
        {reconciliation.map(record => (
          <tr>
            <td>{record.year}</td>
            <td>{record.title}</td>
            <td>{record.owned ? 'OWNED' : 'MISSING'}</td>
            <td>{record.copies.length}</td>
          </tr>
        ))}
      </tbody>
    </table>
  </body>
</html>
